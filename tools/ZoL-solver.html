CONN = {
    "∧": "_and",
    "∨": "_or",
    "⇒": "_implies",
    "⇔": "_iff"
}

EXPR = {}

USED_VARS = []

# MY FUNCTIONS ====================================================================
def _implies(var1: str, var2: str) -> str:
    if len(var1) != len(var2): raise Exception("Size of both variables must be the same!")

    out = ""
    for idx in range(len(var1)):
        if (var1[idx] == "1") and (var2[idx] == "0"): out += "0"
        else: out += "1"
    return out


def _iff(var1: str, var2: str) -> str:
    if len(var1) != len(var2): raise Exception("Size of both variables must be the same!")

    out = ""
    for idx in range(len(var1)):
        if var1[idx] == var2[idx]: out += "1"
        else: out += "0"
    return out


def translate(expr: str) -> str:
    expr = expr.strip()

    # Base Case: Single Variable
    if expr.isalpha() and len(expr) == 1:
        return EXPR[expr]
    
    # Removes outer parantheses if it encloses the whole statement
    if expr.startswith("(") and expr.endswith(")"):
        depth = 0
        encloses_all = True
        
        for i, char in enumerate(expr):
            if char == "(": depth += 1
            elif char == ")": depth -= 1

            if depth == 0 and i < len(expr) - 1: 
                encloses_all = False
                break
        
        if encloses_all:
            expr = expr[1:-1].strip()

    # Find the top-level connector
    depth = 0
    for i, char in enumerate(expr):
        if char == "(": depth += 1
        elif char == ")": depth -= 1

        elif depth == 0 and char in CONN:
            left_operand = expr[:i].strip()
            right_operand = expr[i+1:].strip()

            if not left_operand: raise ValueError("Missing left operand")
            if not right_operand: raise ValueError("Missing right operand")
            
            function = CONN[char]

            return f"{function}(str({translate(left_operand)}), str({translate(right_operand)}))"
    # Handle Negation
    if expr.startswith("¬"):
        operand = expr[1:].strip()
        
        if not operand: raise ValueError("Missing operand after negation")

        return f"_not(str({translate(operand)}))"

    raise ValueError(f"Invalid Expression: {expr}")


def check_argument_validity(statements: list[str]) -> tuple[bool, (int | None)]:
    conclusion = EXPR[statements[-1]]
    num_rows = len(conclusion)
    for row in range(num_rows):
        all_premises_true = all([EXPR[premise][row] == "1" for premise in statements[:-1]])
        if all_premises_true and conclusion[row] == "0": return False, (row + 1)

    return True, None


def check_consistency(premises: list[str]) -> tuple[bool, (int | None)]:
    num_rows = len(EXPR[premises[0]])

    for row in range(num_rows):
        all_premises_true = all([EXPR[premise][row] == "1" for premise in premises])
        if all_premises_true: return True, None

    return False, (row + 1)


def calculate_reliability(statement: str, comp_reliabilities: dict[str, float]) -> float:
    system = EXPR[statement]
    num_rows = len(system)
    reliability = 0.0

    for row in range(num_rows):
        if system[row] == "1":
            row_probability = 1.0

            for var in USED_VARS:
                comp_reliability = comp_reliabilities[var]

                if EXPR[var][row] == "1": row_probability *= comp_reliability
                else: row_probability *= 1 - comp_reliability
            
            reliability += row_probability
    
    return reliability


def calculate_top_event_probability(statement: str, basic_event_probabilities: dict[str, float]) -> float:
    top_event = EXPR[statement]
    num_rows = len(top_event)
    probability = 0.0

    for row in range(num_rows):
        if top_event[row] == "1":
            row_probability = 1.0

            for var in USED_VARS:
                basic_event_probability = basic_event_probabilities[var]

                if EXPR[var][row] == "0": row_probability *= basic_event_probability
                else: row_probability *= 1 - basic_event_probability

            probability += row_probability
        
    return probability


def generate_pdnf(statement: list[str]) -> str:
    results = EXPR[statement[0]]
    num_rows = len(results)
    min_terms = []

    for row in range(num_rows):
        if results[row] == "1":
            literals = []

            for var in USED_VARS:
                if EXPR[var][row] == "1": literals.append(var)
                else: literals.append("¬" + var)

            min_terms.append("(" + " ∧ ".join(literals) + ")")

    if not min_terms:
        return "The proposition is a Fallacy"
    
    return " ∨ ".join(min_terms)


def generate_pcnf(statement: list[str]) -> str:
    results = EXPR[statement[0]]
    num_rows = len(results)
    max_terms = []

    for row in range(num_rows):
        if results[row] == "0":
            literals = []

            for var in USED_VARS:
                if EXPR[var][row] == "0": literals.append(var)
                else: literals.append("¬" + var)
            
            max_terms.append("(" + " ∨ ".join(literals) + ")")

    if not max_terms:
        return "The proposition is a Totality"

    return " ∧ ".join(max_terms)


def gen_truth_table() -> dict[str, str]:
    truth_table = {}
    conversion = {"1": "T", "0": "F"}

    for key, value in EXPR.items():
        truth_table[key] = [conversion[i] for i in value]

    return truth_table

def init_vars(statements: list[str]) -> None:
    """
    ARGUMENTS:
        statements: list[str]
            User input from the UI.
    
    Adds variables to the EXPR dictionary.
    """
    listed_variables = ["P", "Q", "R", "S", "T"]
    used_variables = []

    for var in listed_variables:
        for statement in statements:
            if var in statement:
                used_variables.append(var)
                break
    
    num_vars = len(used_variables)
    for idx, var in enumerate(used_variables):
        num_subsections = 2**(idx+1)        # No of subsections of the representation
        subsection_length = 2**(num_vars - idx - 1)    # Length of each individual subsection
        variable = ""                   
        flag = 1
        for j in range(num_subsections):
            variable += f"{flag}"*subsection_length
            if flag == 1: flag = 0
            else: flag = 1
        EXPR[var] = variable

    for statement in statements:
        EXPR[statement] = str(eval(translate(statement)))
    
    return None

def _and(var1: str, var2: str) -> str:
    """AND operator for two variables of same length."""
    if len(var1) != len(var2): raise Exception("Size of both variables must be the same!")

    out = ""
    for idx in range(len(var1)):
        if (var1[idx] == "1") and (var2[idx] == "1"): out += "1"
        else: out += "0"
    return out

def _or(var1: str, var2: str) -> str:
    """OR operator for two variables of same length."""
    if len(var1) != len(var2): raise Exception("Size of both variables must be the same!")

    out = ""
    for idx in range(len(var1)):
        if (var1[idx] == "0") and (var2[idx] == "0"): out += "0"
        else: out += "1"
    return out

def _not(var1: str) -> str:
    """NOT operator for a variable."""
    out = ""
    for idx in range(len(var1)):
        if var1[idx] == "0": out+= "1"
        else: out += "0"
    return out
